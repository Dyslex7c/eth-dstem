```rust
fn on_stem_ready(&mut self, peer_id: PeerId, frame: &[u8], bodies: PooledTransactions<N::PooledTransaction>, source: TransactionSource, p: f64, send: impl FnOnce(PeerId, alloy_primitives::bytes::BytesMut) -> bool) -> Result<(), &'static str> {
    use alloy_rlp::Decodable;
    let Some((&0x0b, mut data)) = frame.split_first() else { return Err("eth68s code"); };
    let msg = NewPooledTransactionHashes68::decode(&mut data).map_err(|_| "eth68s RLP")?;
    if !data.is_empty() || msg.types.len() != 1 || msg.sizes.len() != 1 || msg.hashes.len() != 1 { return Err("eth68s shape"); }
    if msg.types[0] == 3 || bodies.0.len() != 1 || *bodies.0[0].tx_hash() != msg.hashes[0] { return Err("eth68s body"); }
    let hash = msg.hashes[0].0;
    if let Some(next) = self.stem.handle_inbound_stem(hash, p)? {
        if send(next.into(), frame.into()) { return Ok(()); }
    }
    self.stem.tracker.remove(&hash);
    self.import_transactions(peer_id, bodies, source);
    Ok(())
}
```
